## 📅 2026-01-27 日语学习记录

```json
name: Build & Deploy Vue 2 Project

on:
  push:
    branches:
      - main

permissions:
  contents: write

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest

    steps:
      # 1️⃣ 拉取仓库
      - name: Checkout repository
        uses: actions/checkout@v3.3.0

      # 2️⃣ 设置 Node.js
      - name: Setup Node.js
        uses: actions/setup-node@v3.10.0
        with:
          node-version: '16'

      # 3️⃣ 安装依赖
      - name: Install dependencies
        run: npm install

      # 4️⃣ 构建 Vue 项目
      - name: Build project
        run: npm run build

      # 5️⃣ 禁用 Jekyll
      - name: Disable Jekyll
        run: touch dist/.nojekyll

      # 6️⃣ 删除 dist 下可能存在的 README.md
      - name: Remove README.md if exists
        run: rm -f dist/README.md

      # 7️⃣ SPA history 模式兼容
      - name: Copy index.html to 404.html
        run: cp dist/index.html dist/404.html

      # 8️⃣ 部署到 gh-pages 分支
      - name: Deploy to GitHub Pages
        uses: peaceiris/actions-gh-pages@v3.12.0
        with:
          personal_token: ${{ secrets.GH_PAT }}
          publish_dir: ./dist
          destination_branch: gh-pages
          commit_message: "Deploy Vue 2 project via GitHub Actions"
```
